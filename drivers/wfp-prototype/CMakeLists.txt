cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(orangespace_wfp_prototype LANGUAGES CXX)
endif()
if(WIN32)
  add_library(orangespace-wfp-prototype STATIC network_policy_win.cpp)
  target_compile_features(orangespace-wfp-prototype PUBLIC cxx_std_20)
  target_compile_definitions(orangespace-wfp-prototype PRIVATE _WIN32_WINNT=0x0602)
  target_include_directories(orangespace-wfp-prototype PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(orangespace-wfp-prototype PUBLIC fwpuclnt rpcrt4 ole32 uuid advapi32)
endif()
option(ORANGESPACE_WFP_PROTOTYPE_POLICY_TESTS "Build narrow portable policy test" OFF)
if(ORANGESPACE_WFP_PROTOTYPE_POLICY_TESTS)
  enable_testing()
  add_executable(orangespace-wfp-prototype-policy-tests policy_tests.cpp)
  target_compile_features(orangespace-wfp-prototype-policy-tests PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(orangespace-wfp-prototype-policy-tests PRIVATE /UNDEBUG)
  else()
    target_compile_options(orangespace-wfp-prototype-policy-tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME wfp-prototype-policy COMMAND orangespace-wfp-prototype-policy-tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME wfp-prototype-package-contract
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/package_policy_tests.py)
  if(WIN32)
    # Includes the implementation to exercise private native construction and
    # exact comparison without exposing a production test/configuration seam.
    add_executable(orangespace-wfp-prototype-package-tests package_native_tests.cpp)
    target_compile_features(orangespace-wfp-prototype-package-tests PRIVATE cxx_std_20)
    target_compile_definitions(orangespace-wfp-prototype-package-tests PRIVATE _WIN32_WINNT=0x0602)
    target_link_libraries(orangespace-wfp-prototype-package-tests PRIVATE fwpuclnt rpcrt4 ole32 uuid advapi32)
    if(MSVC)
      target_compile_options(orangespace-wfp-prototype-package-tests PRIVATE /W4 /WX /UNDEBUG)
    else()
      target_compile_options(orangespace-wfp-prototype-package-tests PRIVATE -UNDEBUG)
    endif()
    add_test(NAME wfp-prototype-package-native COMMAND orangespace-wfp-prototype-package-tests)
  endif()
endif()

cmake_minimum_required(VERSION 3.24)
project(OrangeSpaceClient VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(orangespace_client_model INTERFACE)
target_include_directories(orangespace_client_model INTERFACE include)
target_compile_features(orangespace_client_model INTERFACE cxx_std_20)

include(CTest)
if(BUILD_TESTING)
  add_executable(orangespace-file-transfer-tests tests/file_transfer_tests.cpp src/file_transfer_win.cpp)
  target_link_libraries(orangespace-file-transfer-tests PRIVATE orangespace_client_model)
  if(WIN32)
    target_link_libraries(orangespace-file-transfer-tests PRIVATE orangespace_node_session bcrypt)
  endif()
  add_test(NAME file-transfer-policy COMMAND orangespace-file-transfer-tests)
  # These bounded policy executables use assert as their verdict. Keep those
  # assertions active in the Release configuration used by Windows CI too.
  foreach(_ui_policy_target orangespace-file-transfer-tests)
    if(MSVC)
      target_compile_options(${_ui_policy_target} PRIVATE /UNDEBUG /utf-8)
    else()
      target_compile_options(${_ui_policy_target} PRIVATE -UNDEBUG)
    endif()
  endforeach()
  add_executable(orangespace-client-model-tests tests/model_tests.cpp)
  target_link_libraries(orangespace-client-model-tests PRIVATE orangespace_client_model)
  add_test(NAME client-model COMMAND orangespace-client-model-tests)
  add_executable(orangespace-node-session-tests tests/node_session_tests.cpp)
  target_link_libraries(orangespace-node-session-tests PRIVATE orangespace_client_model)
  add_test(NAME node-session COMMAND orangespace-node-session-tests)
  add_executable(orangespace-workspace-auth-tests tests/workspace_auth_tests.cpp)
  target_link_libraries(orangespace-workspace-auth-tests PRIVATE orangespace_client_model)
  add_test(NAME workspace-auth COMMAND orangespace-workspace-auth-tests)
  foreach(test_name node_session_lifecycle node_session_prefix node_session_error credential_store_policy)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${test_name}_tests.cpp")
      add_executable(orangespace-${test_name}-tests "tests/${test_name}_tests.cpp")
      target_link_libraries(orangespace-${test_name}-tests PRIVATE orangespace_client_model)
      add_test(NAME ${test_name} COMMAND orangespace-${test_name}-tests)
    endif()
  endforeach()
  # Keep parser security regressions in both Linux and Windows CTest runs.
  # Allocation hooks and sanitizer instrumentation are separate evidence modes.
  option(ORANGESPACE_JSON_ALLOCATION_AUDIT "Observe parser allocations before release" ON)
  foreach(test_name zeroization transfer inplace_semantic)
    add_executable(orangespace-json-${test_name}-tests "tests/json_${test_name}_tests.cpp")
    target_link_libraries(orangespace-json-${test_name}-tests PRIVATE orangespace_client_model)
    add_test(NAME json-${test_name} COMMAND orangespace-json-${test_name}-tests)
    set_tests_properties(json-${test_name} PROPERTIES TIMEOUT 120 LABELS "json;security")
  endforeach()
  string(CONCAT _orangespace_json_flags "${CMAKE_CXX_FLAGS} "
    "${CMAKE_CXX_FLAGS_DEBUG} ${CMAKE_CXX_FLAGS_RELEASE} "
    "${CMAKE_CXX_FLAGS_RELWITHDEBINFO} ${CMAKE_CXX_FLAGS_MINSIZEREL}")
  if(NOT ORANGESPACE_JSON_ALLOCATION_AUDIT OR _orangespace_json_flags MATCHES "sanitize")
    target_compile_definitions(orangespace-json-zeroization-tests PRIVATE ORANGESPACE_JSON_NO_ALLOC_HOOKS)
    set_tests_properties(json-zeroization PROPERTIES LABELS "json;security;semantic-only")
    message(STATUS "JSON zeroization test: semantic mode; allocation hooks disabled")
  else()
    set_tests_properties(json-zeroization PROPERTIES LABELS "json;security;allocation-audit")
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT WIN32)
    # The positive control deliberately fails if this observer is not enabled.
    target_compile_options(orangespace-json-transfer-tests PRIVATE
      -O0 -fno-inline -fno-elide-constructors -finstrument-functions)
    target_link_options(orangespace-json-transfer-tests PRIVATE -rdynamic)
    target_link_libraries(orangespace-json-transfer-tests PRIVATE ${CMAKE_DL_LIBS})
  endif()
  add_executable(orangespace-sandbox-model-tests tests/sandbox_model_tests.cpp)
  target_link_libraries(orangespace-sandbox-model-tests PRIVATE orangespace_client_model)
  add_test(NAME sandbox-model COMMAND orangespace-sandbox-model-tests)
  if(MSVC)
    target_compile_options(orangespace-sandbox-model-tests PRIVATE /UNDEBUG /utf-8)
  else()
    target_compile_options(orangespace-sandbox-model-tests PRIVATE -UNDEBUG)
  endif()
  add_executable(orangespace-system-setup-tests tests/system_setup_tests.cpp)
  target_include_directories(orangespace-system-setup-tests PRIVATE include)
  add_test(NAME system-setup COMMAND orangespace-system-setup-tests)
  if(MSVC)
    target_compile_options(orangespace-system-setup-tests PRIVATE /UNDEBUG /utf-8 /W4 /permissive-)
  else()
    target_compile_options(orangespace-system-setup-tests PRIVATE -UNDEBUG)
  endif()
  add_executable(orangespace-wsl-channel-tests tests/wsl_channel_tests.cpp)
  target_include_directories(orangespace-wsl-channel-tests PRIVATE include)
  add_test(NAME wsl-channel COMMAND orangespace-wsl-channel-tests)
endif()

if(WIN32)
  # OrangeSpace WSL distribution: import, start from the interactive session,
  # agent channel, shutdown. orangespace-wsl is the acceptance/dev front end.
  add_library(orangespace_wsl_runtime STATIC src/wsl_runtime_win.cpp)
  target_include_directories(orangespace_wsl_runtime PUBLIC include)
  target_compile_features(orangespace_wsl_runtime PUBLIC cxx_std_20)
  target_compile_definitions(orangespace_wsl_runtime PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(orangespace_wsl_runtime PUBLIC advapi32)
  add_executable(orangespace-wsl src/wsl_dev_win.cpp)
  target_link_libraries(orangespace-wsl PRIVATE orangespace_wsl_runtime)
  if(MSVC)
    target_compile_options(orangespace_wsl_runtime PRIVATE /W4 /WX /utf-8)
    target_compile_options(orangespace-wsl PRIVATE /W4 /WX /utf-8)
  endif()
  if(MINGW)
    target_link_options(orangespace-wsl PRIVATE -municode)
  endif()
  # Centre and node sessions, the per-user credential store, and app-private
  # HTTPS through the Rust orangespace-https static library (rustls, only the
  # CA roots this client pins; no system roots, no proxy, no redirects).
  set(ORANGESPACE_HTTPS_LIBRARY "" CACHE FILEPATH "Rust orangespace-https static library for this target")
  if(NOT IS_ABSOLUTE "${ORANGESPACE_HTTPS_LIBRARY}" OR NOT EXISTS "${ORANGESPACE_HTTPS_LIBRARY}")
    message(FATAL_ERROR "Set ORANGESPACE_HTTPS_LIBRARY to the absolute path of orangespace_https.lib (cargo build --release -p orangespace-https)")
  endif()
  # From `cargo rustc -p orangespace-https --crate-type staticlib -- --print native-static-libs`.
  set(ORANGESPACE_HTTPS_SYSTEM_LIBRARIES bcrypt advapi32 kernel32 ntdll userenv ws2_32 dbghelp synchronization)
  add_library(orangespace_node_session STATIC src/node_session_win.cpp src/platform_session_win.cpp src/credential_store_win.cpp)
  target_link_libraries(orangespace_node_session PUBLIC orangespace_client_model
    PRIVATE winhttp crypt32 shell32 ole32 "${ORANGESPACE_HTTPS_LIBRARY}" ${ORANGESPACE_HTTPS_SYSTEM_LIBRARIES})
  target_compile_definitions(orangespace_node_session PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  if(BUILD_TESTING)
    foreach(test_name node_session_lifecycle node_session_prefix node_session_error credential_store_policy)
      if(TARGET orangespace-${test_name}-tests)
        target_link_libraries(orangespace-${test_name}-tests PRIVATE orangespace_node_session)
      endif()
    endforeach()
    add_executable(orangespace-platform-session-tests tests/platform_session_tests.cpp)
    target_link_libraries(orangespace-platform-session-tests PRIVATE orangespace_node_session orangespace_sandbox_engine)
    if(MSVC)
      target_compile_options(orangespace-platform-session-tests PRIVATE /UNDEBUG /utf-8)
    endif()
    add_test(NAME platform-session COMMAND orangespace-platform-session-tests)
    add_executable(orangespace-platform-credential-policy-tests tests/platform_credential_store_policy_tests.cpp)
    target_link_libraries(orangespace-platform-credential-policy-tests PRIVATE orangespace_node_session)
    # Explicit desktop-session fixture (the store refuses session 0); not in ctest.
    add_executable(orangespace-platform-remember-native-tests tests/platform_remember_native_win.cpp)
    target_link_libraries(orangespace-platform-remember-native-tests PRIVATE orangespace_node_session)
    if(MSVC)
      target_compile_options(orangespace-platform-credential-policy-tests PRIVATE /UNDEBUG /utf-8)
      target_compile_options(orangespace-platform-remember-native-tests PRIVATE /UNDEBUG /utf-8)
    endif()
    add_test(NAME platform-credential-policy COMMAND orangespace-platform-credential-policy-tests)
    # Manual acceptance against the real Centre and a node (tunnel replay); not in ctest.
    add_executable(orangespace-tunnel-replay-check tests/tunnel_replay_native_win.cpp)
    target_link_libraries(orangespace-tunnel-replay-check PRIVATE orangespace_node_session bcrypt)
    if(MINGW)
      target_link_options(orangespace-tunnel-replay-check PRIVATE -municode)
    endif()
    if(MSVC)
      target_compile_options(orangespace-tunnel-replay-check PRIVATE /W4 /permissive- /utf-8)
    endif()
    # Manual check of the Centre clock offset against the real Centre; not in ctest.
    add_executable(orangespace-centre-clock-check tests/centre_clock_native_win.cpp)
    target_link_libraries(orangespace-centre-clock-check PRIVATE orangespace_node_session)
    if(MINGW)
      target_link_options(orangespace-centre-clock-check PRIVATE -municode)
    endif()
    if(MSVC)
      target_compile_options(orangespace-centre-clock-check PRIVATE /W4 /permissive- /utf-8)
    endif()
    # Explicit headless fixture; native execution is opt-in, never on startup.
    add_executable(orangespace-file-directory-native-tests WIN32
      tests/file_transfer_directory_native_win.cpp src/file_transfer_win.cpp)
    target_include_directories(orangespace-file-directory-native-tests PRIVATE include)
    target_compile_definitions(orangespace-file-directory-native-tests PRIVATE
      ORANGESPACE_FILE_LOCAL_IO_ONLY UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(orangespace-file-directory-native-tests PRIVATE bcrypt shell32 ole32 uuid advapi32)
    if(MINGW)
      target_link_options(orangespace-file-directory-native-tests PRIVATE -municode)
    endif()
    # Explicit fixture runners; never count session-0 or absent HTTPS as acceptance.
    add_executable(orangespace-credential-session-native-tests tests/credential_session_native_win.cpp)
    target_link_libraries(orangespace-credential-session-native-tests PRIVATE orangespace_node_session)
    add_executable(orangespace-credential-store-native-tests tests/credential_store_native_win.cpp src/node_session_win.cpp src/platform_session_win.cpp)
    target_link_libraries(orangespace-credential-store-native-tests PRIVATE orangespace_client_model winhttp crypt32 shell32 ole32
      "${ORANGESPACE_HTTPS_LIBRARY}" ${ORANGESPACE_HTTPS_SYSTEM_LIBRARIES})
    target_compile_definitions(orangespace-credential-store-native-tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    if(MINGW)
      target_link_options(orangespace-credential-session-native-tests PRIVATE -municode)
    endif()
  endif()
  # Sandboxes on node leases: one worker per sandbox (own node session and
  # tunnel, renewals that keep its kernel permit alive). Used by the window and
  # by the acceptance front end orangespace-sandbox.
  add_library(orangespace_sandbox_engine STATIC src/sandbox_engine_win.cpp)
  target_include_directories(orangespace_sandbox_engine PUBLIC include)
  target_link_libraries(orangespace_sandbox_engine PUBLIC orangespace_node_session orangespace_wsl_runtime PRIVATE bcrypt)
  if(MSVC)
    target_compile_options(orangespace_sandbox_engine PRIVATE /W4 /WX /permissive- /utf-8)
  endif()
  # Node personal files (FileClient) and a sandbox's Downloads (GuestFiles),
  # used by the files window of OrangeSpace.exe.
  add_library(orangespace_file_client STATIC src/file_transfer_win.cpp src/sandbox_files_win.cpp)
  target_link_libraries(orangespace_file_client PUBLIC orangespace_node_session orangespace_wsl_runtime PRIVATE winhttp bcrypt)
  if(MSVC)
    target_compile_options(orangespace_file_client PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_executable(orangespace-sandbox src/sandbox_session_win.cpp)
  target_link_libraries(orangespace-sandbox PRIVATE orangespace_sandbox_engine)
  if(MINGW)
    target_link_options(orangespace-sandbox PRIVATE -municode)
  endif()
  if(MSVC)
    target_compile_options(orangespace-sandbox PRIVATE /W4 /permissive- /utf-8)
  endif()
  if(TARGET orangespace-node-session-tests)
    target_link_libraries(orangespace-node-session-tests PRIVATE orangespace_node_session)
    add_executable(orangespace-node-tls-rejection-tests tests/node_tls_rejection_win.cpp)
    target_link_libraries(orangespace-node-tls-rejection-tests PRIVATE orangespace_node_session)
    add_test(NAME node-tls-rejection COMMAND orangespace-node-tls-rejection-tests)
    set_tests_properties(node-tls-rejection PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
    if(MINGW)
      target_link_options(orangespace-node-tls-rejection-tests PRIVATE -municode)
    endif()
  endif()
  if(MSVC)
    target_compile_options(orangespace_node_session PRIVATE /W4 /permissive- /utf-8)
  endif()
  enable_language(RC)
  add_executable(OrangeSpace WIN32 src/app_win.cpp src/files_window_win.cpp src/system_setup_win.cpp src/brand.rc)
  if(MSVC)
    # Let MSVC merge and embed the manifest once. An RC-embedded copy would
    # collide with the linker's generated RT_MANIFEST resource (CVT1100).
    target_sources(OrangeSpace PRIVATE src/app.manifest)
    if(BUILD_TESTING)
      if(CMAKE_MT)
        set(ORANGESPACE_MANIFEST_TOOL "${CMAKE_MT}")
      else()
        # Visual Studio may locate the SDK internally while CMAKE_MT is unset.
        get_filename_component(_orangespace_rc_directory "${CMAKE_RC_COMPILER}" DIRECTORY)
        find_program(ORANGESPACE_MANIFEST_TOOL NAMES mt.exe
          HINTS "${_orangespace_rc_directory}"
          PATHS "[HKEY_LOCAL_MACHINE/SOFTWARE/Microsoft/Windows Kits/Installed Roots;KitsRoot10]/bin/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/x64"
          REGISTRY_VIEW HOST REQUIRED)
      endif()
      add_test(NAME windows-application-manifest COMMAND ${CMAKE_COMMAND}
        "-DMT_EXECUTABLE=${ORANGESPACE_MANIFEST_TOOL}"
        "-DAPPLICATION=$<TARGET_FILE:OrangeSpace>"
        "-DOUTPUT_FILE=${CMAKE_CURRENT_BINARY_DIR}/verified-$<CONFIG>.manifest"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_manifest.cmake")
    endif()
  else()
    target_sources(OrangeSpace PRIVATE src/app.rc)
  endif()
  target_link_libraries(OrangeSpace PRIVATE orangespace_sandbox_engine orangespace_file_client comctl32 shell32 user32 gdi32 ole32 bcrypt)
  target_compile_definitions(OrangeSpace PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_include_directories(OrangeSpace PRIVATE src)
  if(TARGET orangespace-client-model-tests)
    target_link_libraries(orangespace-client-model-tests PRIVATE shell32)
  endif()
  if(MSVC)
    target_compile_options(OrangeSpace PRIVATE /W4 /WX /permissive- /utf-8)
    if(TARGET orangespace-client-model-tests)
      target_compile_options(orangespace-client-model-tests PRIVATE /W4 /permissive- /utf-8)
    endif()
  endif()
  if(MINGW)
    target_link_options(OrangeSpace PRIVATE -municode -Wl,--subsystem,windows)
  endif()
endif()

cmake_minimum_required(VERSION 3.24)
project(OrangeSpace VERSION 0.1.0 LANGUAGES CXX)
set(ORANGESPACE_DEVELOPMENT_INSTALL_ID "" CACHE STRING "Immutable isolated development install suffix")
string(LENGTH "${ORANGESPACE_DEVELOPMENT_INSTALL_ID}" _install_id_length)
if(NOT ORANGESPACE_DEVELOPMENT_INSTALL_ID STREQUAL "")
    if(NOT ORANGESPACE_DEVELOPMENT_INSTALL_ID MATCHES "^[a-z0-9-]+$" OR _install_id_length GREATER 24)
        message(FATAL_ERROR "Development install id must be 1..24 lowercase ASCII alnum/hyphen")
    endif()
endif()
add_compile_definitions(ORANGESPACE_DEVELOPMENT_INSTALL_ID="${ORANGESPACE_DEVELOPMENT_INSTALL_ID}")
include(CTest)
add_subdirectory(apps/windows-client)
if(WIN32)
    add_subdirectory(drivers/wfp-baseline)
endif()
